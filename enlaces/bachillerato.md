  <script>
    (() => {
      'use strict';
      const search = document.getElementById('resource-search');
      const select = document.getElementById('subject-select');
      const rows = Array.from(document.querySelectorAll('#resource-rows .resource-row'));
      const chips = Array.from(document.querySelectorAll('[data-topic]'));
      const counter = document.getElementById('result-count');
      const empty = document.getElementById('empty-state');
      const clear = document.getElementById('clear-filters');
      const resetFromEmpty = document.getElementById('empty-reset');
      const normalizeText = text => String(text).normalize('NFD').replace(/[\u0300-\u036f]/g, '').toLowerCase();
      const searchable = rows.map(row => normalizeText(row.textContent));
      let topic = 'all';
      function render() {
        const query = normalizeText(search.value.trim());
        let shown = 0;
        rows.forEach((row, i) => {
          const categories = row.dataset.categories.split(' ');
          const visible = (topic === 'all' || categories.includes(topic)) && (!query || searchable[i].includes(query));
          row.hidden = !visible;
          if (visible) shown++;
        });
        counter.innerHTML = `Mostrando <strong>${shown}</strong> de <strong>${rows.length}</strong> recursos`;
        empty.hidden = shown !== 0;
        clear.hidden = topic === 'all' && !query;
        chips.forEach(chip => {
          const selected = chip.dataset.topic === topic;
          chip.classList.toggle('active', selected);
          chip.setAttribute('aria-pressed', String(selected));
        });
        select.value = topic;
      }
      function reset() { search.value = ''; topic = 'all'; render(); search.focus(); }
      search.addEventListener('input', render);
      select.addEventListener('change', event => { topic = event.target.value; render(); });
      chips.forEach(chip => chip.addEventListener('click', () => { topic = chip.dataset.topic; render(); }));
      clear.addEventListener('click', reset);
      resetFromEmpty.addEventListener('click', reset);
      render();
    })();
  </script>
